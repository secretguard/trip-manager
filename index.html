Google Drive Sync

Data is saved to your Google Drive via Google Apps Script. No tokens, no expiry.

One-time setup:
1. Go to script.google.com → New project → paste the GAS code below → Save
2. Click Deploy → New deployment → Web app
3. Execute as: Me · Who has access: Anyone → Deploy → Allow
4. Copy the Web App URL → paste into GAS_URL in index.html → re-upload to GitHub

GAS Script — paste into script.google.com
const FILE_NAME = 'trips_backup.json';
const FOLDER_NAME = 'Trip Manager';

function getFolder() {
  const folders = DriveApp.getFoldersByName(FOLDER_NAME);
  if (folders.hasNext()) return folders.next();
  return DriveApp.createFolder(FOLDER_NAME);
}
function findFile() {
  const folder = getFolder();
  const files = folder.getFilesByName(FILE_NAME);
  return files.hasNext() ? files.next() : null;
}
function doGet(e) {
  const action = e?.parameter?.action || 'load';
  if (action === 'ping') return json({ status: 'ok' });
  if (action === 'load') {
    const file = findFile();
    if (!file) return json({ status: 'ok', data: null });
    return json({ status: 'ok', data: JSON.parse(file.getBlob().getDataAsString()) });
  }
  return json({ status: 'error', error: 'Unknown action' });
}
function doPost(e) {
  try {
    const payload = JSON.parse(e.postData.contents);
    if (payload.action === 'save') {
      const c = JSON.stringify(payload.data, null, 2);
      const f = findFile();
      if (f) f.setContent(c);
      else getFolder().createFile(FILE_NAME, c, MimeType.PLAIN_TEXT);
      return json({ status: 'ok' });
    }
  } catch(err) { return json({ status: 'error', error: err.message }); }
}
function json(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}
